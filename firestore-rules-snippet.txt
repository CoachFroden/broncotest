// IKKE deploy denne filen som hele rules-filen, fordi prosjektet også brukes av andre apper.
// Bronco-resultater er trenerdata og skal kun være tilgjengelige for brukere med role == "coach".

function isBroncoCoach() {
  return request.auth != null
    && exists(/databases/$(database)/documents/users/$(request.auth.uid))
    && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "coach";
}

match /broncoTests/{testId} {
  allow read, create, update, delete: if isBroncoCoach();
}

match /broncoConfig/{docId} {
  allow read, create, update: if isBroncoCoach();
  allow delete: if false;
}
